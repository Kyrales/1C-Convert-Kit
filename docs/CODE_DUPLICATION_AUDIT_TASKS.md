# В V8ToolWrapper добавлен метод:
def _append_ib_credentials(self, cmd: list[str]) -> None:
    """
    Добавление учетных данных к команде 1С.
    
    Args:
        cmd: Список аргументов команды (модифицируется in-place)
    """
    ib_user = self.env_vars.get('V8_IB_USER', '')
    ib_pwd = self.env_vars.get('V8_IB_PWD', '')
    if ib_user:
        cmd.append(f'/N{ib_user}')
    if ib_pwd:
        cmd.append(f'/P{ib_pwd}')
```

# Создан src/utils/time_utils.py:
"""Утилиты для работы со временем."""

def format_duration(seconds: float) -> str:
    """
    Форматирование длительности в читаемый вид.
    
    Args:
        seconds: Длительность в секундах
    
    Returns:
        Отформатированная строка (например: "1ч 23м 45с")
    """
    if seconds < 0:
        return "0с"
    
    days = int(seconds // 86400)
    hours = int((seconds % 86400) // 3600)
    minutes = int((seconds % 3600) // 60)
    secs = int(seconds % 60)
    
    parts = []
    if days > 0:
        parts.append(f"{days}д")
    if hours > 0:
        parts.append(f"{hours}ч")
    if minutes > 0:
        parts.append(f"{minutes}м")
    if secs > 0 or not parts:
        parts.append(f"{secs}с")
    
    return " ".join(parts)
```
